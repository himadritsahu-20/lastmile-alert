function showPage(page) {
    // Hide all pages
    document.querySelectorAll('[id^="page-"]').forEach(el => el.classList.add('hidden'));
    document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));

    // Show selected page
    const pageEl = document.getElementById('page-' + page);
    if (pageEl) pageEl.classList.remove('hidden');

    // Activate nav
    const navEl = document.getElementById('nav-' + page);
    if (navEl) navEl.classList.add('active');

    // Update title
    const titles = {
        'dashboard': 'Dashboard',
        'alert-input': 'Create Alert',
        'plain-language': 'Plain Language Processor',
        'translator': 'Multi-Language Translator',
        'visual': 'Visual Generator',
        'sms': 'SMS & Low-Bandwidth Simulator',
        'tracker': 'Delivery Tracker'
    };
    document.getElementById('page-title').textContent = titles[page] || page;

    // Special init
    if (page === 'dashboard') initCharts();
    if (page === 'tracker') renderRecipientTable();
    if (page === 'visual') generateVisual();
}

function selectType(btn, type) {
    document.querySelectorAll('.type-btn').forEach(b => {
        b.classList.remove('border-blue-500');
        b.classList.add('border-transparent');
    });
    btn.classList.add('border-blue-500');
    btn.classList.remove('border-transparent');
    AppState.currentType = type;
}

function loadSampleAlert(type) {
    const sample = SAMPLE_ALERTS[type];
    if (!sample) return;
    
    AppState.currentAlert = sample;
    AppState.currentType = type;
    
    const textarea = document.getElementById('official-alert');
    const areaInput = document.getElementById('affected-area');
    if (textarea) textarea.value = sample.text;
    if (areaInput) areaInput.value = sample.area;

    updateCharCount();
    showToast('Sample alert loaded! Click Process Alert.');
}

function updateCharCount() {
    const textarea = document.getElementById('official-alert');
    const counter = document.getElementById('char-count');
    if (textarea && counter) {
        counter.textContent = textarea.value.length + ' characters';
    }
}

document.addEventListener('DOMContentLoaded', () => {
    const textarea = document.getElementById('official-alert');
    if (textarea) textarea.addEventListener('input', updateCharCount);
});

function processAlert() {
    const alertText = document.getElementById('official-alert');
    const areaInput = document.getElementById('affected-area');
    const severity = document.getElementById('severity');
    
    if (!alertText || !alertText.value.trim()) {
        showToast('Please enter alert text first!', 'error');
        return;
    }

    AppState.currentAlert = {
        type: AppState.currentType,
        severity: severity ? severity.value : 'critical',
        area: areaInput ? areaInput.value : 'Unknown',
        text: alertText.value,
        simplified: SAMPLE_ALERTS[AppState.currentType]?.simplified || simplifyText(alertText.value)
    };

    // Update other pages
    updatePlainLanguagePage();
    updateTranslatorPage();

    showToast('Alert processed! Check Plain Language tab.');
    setTimeout(() => showPage('plain-language'), 1500);
}

function simplifyText(text) {
    return `🚨 EMERGENCY ALERT - ACTION REQUIRED!

${text.substring(0, 200)}...

✅ WHAT TO DO:
• Follow official instructions
• Contact emergency services
• Stay tuned for updates

📞 EMERGENCY: 112`;
}

function updatePlainLanguagePage() {
    if (!AppState.currentAlert) return;
    
    const origDisplay = document.getElementById('original-text-display');
    const simpDisplay = document.getElementById('simplified-text-display');
    
    if (origDisplay) origDisplay.textContent = AppState.currentAlert.text;
    if (simpDisplay) simpDisplay.innerHTML = AppState.currentAlert.simplified?.replace(/\n/g, '<br>') || '';

    // Update key info
    const typeData = {
        flood: {what: 'Flood / Rising Water', where: 'Zone A, District 5', when: 'Next 4 hours', action: 'EVACUATE NOW'},
        cyclone: {what: 'Cyclone / Storm', where: 'Coastal Region', when: '36-48 hours', action: 'PREPARE & SHELTER'},
        fire: {what: 'Forest Fire', where: 'District B, North', when: 'Monitor every 30 min', action: 'BE READY TO LEAVE'},
        earthquake: {what: 'Earthquake', where: 'Affected Zone', when: 'Immediate', action: 'DUCK COVER HOLD'}
    };
    
    const info = typeData[AppState.currentAlert.type] || typeData.flood;
    
    const els = ['what','where','when','action'];
    els.forEach(key => {
        const el = document.getElementById('key-' + key);
        if (el) el.textContent = info[key];
    });
}

function copyText(elementId) {
    const el = document.getElementById(elementId);
    if (el) {
        navigator.clipboard.writeText(el.innerText);
        showToast('Text copied to clipboard!');
    }
}

function updateTranslatorPage() {
    if (!AppState.currentAlert) return;
    const sourceEl = document.getElementById('source-for-translation');
    if (sourceEl) {
        sourceEl.textContent = AppState.currentAlert.simplified || AppState.currentAlert.text;
    }
}

function translateTo(language) {
    if (!AppState.currentAlert) {
        loadSampleAlert('flood');
        setTimeout(() => translateTo(language), 200);
        return;
    }
    
    const type = AppState.currentAlert.type;
    const el = document.getElementById('trans-' + language);
    if (!el) return;
    
    el.innerHTML = '<span class="text-yellow-400">Translating...</span>';
    
    setTimeout(() => {
        const trans = TRANSLATIONS[language]?.[type] || 
            TRANSLATIONS[language]?.flood || 
            `[${language.toUpperCase()}] Alert: ${AppState.currentAlert.simplified?.substring(0, 100)}...`;
        
        el.innerHTML = trans.substring(0, 150).replace(/\n/g, '<br>') + '...';
        AppState.translations[language] = trans;
        showToast(`Translated to ${language}!`);
    }, 800);
}

function translateAll() {
    if (!AppState.currentAlert) {
        loadSampleAlert('flood');
        setTimeout(translateAll, 200);
        return;
    }
    
    const langs = ['hindi','bengali','tamil','telugu','spanish','french','arabic','malay'];
    langs.forEach((lang, idx) => {
        setTimeout(() => translateTo(lang), idx * 300);
    });
    showToast('Translating to all languages...');
}

function generateVisual() {
    const canvas = document.getElementById('visual-canvas');
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    
    const alert = AppState.currentAlert || SAMPLE_ALERTS.flood;
    const type = alert.type || 'flood';
    
    const configs = {
        flood: { color: '#0066CC', bg: '#001133', icon: '🌊', title: 'FLOOD WARNING', action: 'EVACUATE NOW' },
        cyclone: { color: '#FF6600', bg: '#1A0A00', icon: '🌀', title: 'CYCLONE ALERT', action: 'SEEK SHELTER' },
        fire: { color: '#FF3300', bg: '#1A0500', icon: '🔥', title: 'FIRE ALERT', action: 'STAY READY' },
        earthquake: { color: '#8B4513', bg: '#0D0A00', icon: '🏔️', title: 'EARTHQUAKE', action: 'DROP-COVER-HOLD' }
    };
    
    const config = configs[type] || configs.flood;
    
    // Background
    ctx.fillStyle = config.bg;
    ctx.fillRect(0, 0, 400, 400);
    
    // Header bar
    ctx.fillStyle = config.color;
    ctx.fillRect(0, 0, 400, 80);
    
    // Title
    ctx.fillStyle = '#FFFFFF';
    ctx.font = 'bold 22px Inter, Arial';
    ctx.textAlign = 'center';
    ctx.fillText(config.title, 200, 50);
    
    // Icon (emoji representation)
    ctx.font = '80px Arial';
    ctx.fillText(config.icon, 200, 175);
    
    // Severity badge
    ctx.fillStyle = '#FF3A3A';
    ctx.beginPath();
    ctx.roundRect(100, 200, 200, 40, 20);
    ctx.fill();
    ctx.fillStyle = '#FFFFFF';
    ctx.font = 'bold 16px Inter, Arial';
    ctx.fillText('⚠️ CRITICAL DANGER', 200, 225);
    
    // Action
    ctx.fillStyle = '#FFFFFF';
    ctx.font = 'bold 24px Inter, Arial';
    ctx.fillText(config.action, 200, 285);
    
    // Info boxes
    const boxData = [
        { label: 'LOCATION', value: alert.area?.substring(0, 15) || 'Zone A' },
        { label: 'CALL', value: '📞 112' }
    ];
    
    boxData.forEach((box, i) => {
        const x = 20 + i * 200;
        ctx.strokeStyle = config.color;
        ctx.lineWidth = 2;
        ctx.strokeRect(x, 310, 180, 60);
        ctx.fillStyle = '#888888';
        ctx.font = '11px Inter, Arial';
        ctx.textAlign = 'center';
        ctx.fillText(box.label, x + 90, 330);
        ctx.fillStyle = '#FFFFFF';
        ctx.font = 'bold 14px Inter, Arial';
        ctx.fillText(box.value, x + 90, 355);
    });
    
    // Footer
    ctx.fillStyle = config.color;
    ctx.fillRect(0, 385, 400, 15);
    ctx.fillStyle = '#FFFFFF';
    ctx.font = '10px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('OFFICIAL GOVERNMENT ALERT - LastMile Alert System', 200, 396);
}

function setNetwork(type) {
    AppState.currentNetwork = type;
    
    const configs = {
        full: { name: 'Full Internet', speed: 'Speed: Unlimited', signal: '▲▲▲▲', signalColor: 'text-green-400', delay: 0 },
        '3g': { name: '3G Network', speed: 'Speed: ~1 Mbps', signal: '▲▲▲░', signalColor: 'text-yellow-400', delay: 2000 },
        '2g': { name: '2G Network', speed: 'Speed: ~64 Kbps', signal: '▲▲░░', signalColor: 'text-orange-400', delay: 5000 },
        sms: { name: 'SMS Only', speed: 'Speed: SMS mode', signal: '▲░░░', signalColor: 'text-red-400', delay: 3000 },
        offline: { name: 'Offline Relay', speed: 'Speed: Community relay', signal: '░░░░', signalColor: 'text-gray-500', delay: 10000 }
    };
    
    const config = configs[type] || configs.full;
    
    const networkEl = document.getElementById('current-network');
    const speedEl = document.getElementById('network-speed');
    const signalEl = document.getElementById('signal-display');
    
    if (networkEl) networkEl.textContent = config.name;
    if (speedEl) speedEl.textContent = config.speed;
    if (signalEl) {
        signalEl.textContent = config.signal;
        signalEl.className = 'text-xs ' + config.signalColor;
    }
    
    showToast(`Network set to: ${config.name}`);
}

function simulateSend() {
    const delays = { full: 500, '3g': 2000, '2g': 5000, sms: 3000, offline: 10000 };
    const delay = delays[AppState.currentNetwork] || 500;
    
    const timeEl = document.getElementById('delivery-time');
    if (timeEl) timeEl.textContent = 'Sending...';
    
    const phoneMsg = document.getElementById('phone-message');
    if (phoneMsg) phoneMsg.classList.add('opacity-50');
    
    setTimeout(() => {
        if (timeEl) timeEl.textContent = `Delivered after ${delay/1000}s`;
        if (phoneMsg) phoneMsg.classList.remove('opacity-50');
        showToast(`Message delivered! (${delay/1000}s delay)`);
    }, delay);
}

function buildSMS() {
    const code = document.getElementById('sms-code')?.value || '';
    const zone = document.getElementById('sms-zone')?.value || '';
    const action = document.getElementById('sms-action')?.value || '';
    const contact = document.getElementById('sms-contact')?.value || '';
    
    const sms = `${code}|${zone}|${action}|${contact}|#${Date.now().toString().slice(-6)}`;
    const smsEl = document.getElementById('generated-sms');
    const countEl = document.getElementById('sms-char-count');
    
    if (smsEl) smsEl.textContent = sms;
    if (countEl) countEl.textContent = `${sms.length}/160 characters`;
    
    showToast('SMS format built!');
}

function renderRecipientTable() {
    const tbody = document.getElementById('recipient-table');
    if (!tbody) return;
    
    const statusConfig = {
        'acknowledged': { color: 'text-green-400', bg: 'bg-green-900', dot: 'bg-green-400' },
        'read': { color: 'text-blue-400', bg: 'bg-blue-900', dot: 'bg-blue-400' },
        'delivered': { color: 'text-yellow-400', bg: 'bg-yellow-900', dot: 'bg-yellow-400' },
        'failed': { color: 'text-red-400', bg: 'bg-red-900', dot: 'bg-red-400' }
    };
    
    tbody.innerHTML = RECIPIENTS.map(r => {
        const s = statusConfig[r.status] || statusConfig.delivered;
        return `
        <tr class="border-b border-gray-800 hover:bg-gray-800 transition-colors">
            <td class="py-3 px-3">
                <div class="flex items-center gap-2">
                    <span class="status-dot ${s.dot}"></span>
                    <div>
                        <div class="font-semibold text-sm">${r.name}</div>
                        <div class="text-xs text-gray-500">${r.id}</div>
                    </div>
                </div>
            </td>
            <td class="py-3 px-3 text-sm">Zone ${r.zone}</td>
            <td class="py-3 px-3 text-sm">${r.channel}</td>
            <td class="py-3 px-3 text-sm text-gray-400">${r.time}</td>
            <td class="py-3 px-3">
                <span class="${s.bg} ${s.color} text-xs px-2 py-1 rounded-full capitalize">${r.status}</span>
            </td>
            <td class="py-3 px-3">
                <span class="${r.ack === 'Yes' ? 'text-green-400' : 'text-gray-500'} text-sm font-semibold">${r.ack}</span>
            </td>
        </tr>`;
    }).join('');
}

function exportCSV() {
    const headers = ['ID', 'Name', 'Zone', 'Channel', 'Time', 'Status', 'Acknowledged'];
    const rows = RECIPIENTS.map(r => 
        [r.id, r.name, r.zone, r.channel, r.time, r.status, r.ack].join(',')
    );
    const csv = [headers.join(','), ...rows].join('\n');
    
    const blob = new Blob([csv], { type: 'text/csv' });
    const link = document.createElement('a');
    link.href = URL.createObjectURL(blob);
    link.download = 'delivery_report.csv';
    link.click();
    showToast('CSV exported!');
}

function showToast(message, type = 'success') {
    const toast = document.getElementById('toast');
    if (!toast) return;
    
    toast.textContent = (type === 'error' ? '❌ ' : '✅ ') + message;
    toast.className = `fixed bottom-4 right-4 ${type === 'error' ? 'bg-red-600' : 'bg-green-600'} text-white px-6 py-3 rounded-xl slide-in z-50`;
    toast.classList.remove('hidden');
    
    setTimeout(() => toast.classList.add('hidden'), 3000);
}
